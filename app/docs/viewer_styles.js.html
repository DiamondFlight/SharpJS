/** Add the stylesheet to `doc` once. */
export function injectStyles(doc = typeof document !== 'undefined' ? document : null) {
  if (!doc || injected.has(doc) || doc.querySelector('style[data-sharpjs]')) return;
  const style = doc.createElement('style');
  style.setAttribute('data-sharpjs', '');
  style.textContent = SHARPJS_CSS;
  (doc.head || doc.documentElement).appendChild(style);
  injected.add(doc);
}
