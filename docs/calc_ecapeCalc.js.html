/**
 * Estimate saturation vapor pressure from temperature using the module's polynomial fit.
 * @param {number} temp Temperature (in °C).
 * @returns {number} Saturation vapor pressure (in hPa), or -9999 for an invalid temperature.
 */
function vappres(temp) {
  if (!qc(temp)) return -9999;

  var pol = temp * (1.1112018E-17 + temp * (-3.0994571E-20));
  pol = temp * (2.1874425E-13 + temp * (-1.789232E-15 + pol));
  pol = temp * (4.3884180E-09 + temp * (-2.988388E-11 + pol));
  pol = temp * (7.8736169E-05 + temp * (-6.111796E-07 + pol));
  pol = 0.99999683E-00 + temp * (-9.082695E-03 + pol);
  pol = pol * pol;
  pol = pol * pol;
  return 6.1078 / (pol * pol);
}

/**
 * Calculate saturation mixing ratio from pressure and temperature.
 * @param {number} pres Ambient pressure (in hPa).
 * @param {number} temp Temperature (in °C).
 * @returns {number} Saturation mixing ratio (in g/kg), or -9999 for invalid input.
 */
function mixratio(pres, temp) {
  if (!qc(pres) || !qc(temp)) return -9999;

  var x = 0.02 * (temp - 12.5 + 7500.0 / pres);
  var wfw = 1.0 + 0.0000045 * pres + 0.0014 * x * x;
  var fwesw = wfw * vappres(temp);
  return 621.97 * (fwesw / (pres - fwesw));
}

/**
 * Linearly interpolate between two numeric values.
 * @param {number} a Starting value
 * @param {number} b Ending value
 * @param {number} t Interpolation fraction.
 * @returns {number} Interpolated value
 */
function Lerp(a, b, t){
  return a + (b-a)*t;
}

/**
 * Convert speed and direction angle to Cartesian vector components.
 * @param {number} Spd Vector magnitude (in kt).
 * @param {number} Tta Direction angle (in degrees).
 * @returns {{x: number, y: number}} Cartesian components (in kt).
 */
export function PolarToCartesianDeg(Spd, Tta){
  return {x: Spd * Math.cos(DegToRad(Tta)), y: Spd * Math.sin(DegToRad(Tta))};
}

/**
 * Convert Cartesian vector components to magnitude and direction angle.
 * @param {{x: number, y: number}} T Vector components (in kt).
 * @returns {{r: number, theta: number}} Magnitude (in kt) and angle (in degrees).
 */
export function CartesianToPolarDeg(T){
  return {r: Math.sqrt(Math.pow(T.x, 2) + Math.pow(T.y, 2)), theta: RadToDeg(Math.atan2(T.y, T.x))};
}

/** Lerp every field between rows `Pr` and `Rr`; wind (4 = dir, 5 = speed) via u/v components. */
function LerpSoundingSlice(Pr, Rr, LerpFactor){
  let Slice = {};
  for (let x = 0; x < Pr.length; x++){
    if (x == 4 || x == 5){
      if (x == 5){
        continue;
      }

      let Cartesian = PolarToCartesianDeg(Pr[5], Pr[4]);
      let Cartesian2 = PolarToCartesianDeg(Rr[5], Rr[4]);

      let TrueCart = {
        x: Lerp(Cartesian.x, Cartesian2.x, LerpFactor),
        y: Lerp(Cartesian.y, Cartesian2.y, LerpFactor),
      };

      let TruePolarInterp = CartesianToPolarDeg(TrueCart);

      if (TruePolarInterp.theta < 0){
        TruePolarInterp.theta = (TruePolarInterp.theta + 360) % 360;
      }

      Slice[4] = TruePolarInterp.theta;
      Slice[5] = TruePolarInterp.r;
    } else {
      Slice[x] = Lerp(Pr[x], Rr[x], LerpFactor);
    }
  }
  return Slice;
}

/** entrainment rate in 1/m
 * 
 * @param {*} cape in j/kg
 * @param {*} ecape in j/kg
 * @param {*} ncape in j/kg
 * @param {*} vsr idk
 * @param {*} sch storm coloumn height
 * @returns {number} entrainment rate (M^-1)
 */
export function entrainment_rate(cape, ecape, ncape, vsr, sch){
    const eat = ecape/cape;
    const nt = ncape/cape;
    const vsrt = vsr/Math.sqrt(2 * cape);
    const et = eat - Math.pow(vsrt, 2);

    return (2 * (1 - et)/(et + nt))/sch;
}

/**
 * Evaluate a step function, using x2 as the value exactly at the threshold.
 * @param {number} x1 Signed distance from the threshold
 * @param {number} x2 Value at the threshold
 * @returns {number} 0, x2, or 1 according to the sign of x1
 */
function heaviside(x1, x2) {
  if (x1 < 0) {
    return 0;
  } else if (x1 === 0) {
    return x2;
  } else {
    return 1;
  }
}

/**
 * Calculate the fractional mixed-phase ice content between two temperature limits.
 * @param {number} T Temperature (in K).
 * @param {number} T1 Warm mixed-phase limit (in K).
 * @param {number} T2 Cold mixed-phase limit (in K).
 * @returns {number} Ice fraction from 0 to 1 (in unitless).
 */
function omega(T, T1, T2){
    return ((T - T1)/(T2 - T1)) * heaviside((T - T1)/(T2 - T1), 1) * heaviside((1 - (T - T1)/(T2-T1)),1) + heaviside(-(1 - (T - T1)/(T2-T1)),1);
}

/**
 * Calculate the temperature derivative of the mixed-phase ice fraction.
 * @param {number} T Temperature (in K).
 * @param {number} T1 Warm mixed-phase limit (in K).
 * @param {number} T2 Cold mixed-phase limit (in K).
 * @returns {number} Ice-fraction derivative (in K⁻¹).
 */
function domega(T, T1, T2){
    return (heaviside(T1-T,1) - heaviside(T2-T,1))/(T2-T1);
}

/** Normal Cape
 * 
 * @param {*} T3 C array
 * @param {*} p0 Pa array
 * @param {*} q0 kg/kg array
 * @param {*} z0 m array
 * @param {*} Snding Sounding
 * @param {*} T1 kevin
 * @param {*} T2 kevin
 * @param {*} LFC m
 * @param {*} EL m
 */
export function compute_NCAPE(T0, p0, q0, z0, Snding, T1, T2, LFC, EL){
    // Previous resampling approach rebuilt a 100-m grid from the sounding
    // instead of using the supplied temperature, moisture, pressure, and height arrays.
    // const surfaceHeight = z0[0];
    // for (let height = surfaceHeight; height < EndSlice.H; height += 100) { ... }
    const MSE0 = T0.map((temperature, index) =>
        Constants.c_pd * temperature + Constants.xlv * q0[index] + Constants.g * z0[index]);
    const rsat = T0.map((temperature, index) => r_sat(temperature, p0[index], 0, T1, T2));
    const qsat = rsat.map(mixingRatio => (1 - mixingRatio) * mixingRatio);
    const MSE0_star = T0.map((temperature, index) =>
        Constants.c_pd * temperature + Constants.xlv * qsat[index] + Constants.g * z0[index]);
    const MSE0_bar = new Array(MSE0.length).fill(0);
    MSE0_bar[0] = MSE0[0];

    for (let iz = 1; iz < MSE0_bar.length; iz++){
        let cumulativeMSE = 0;

        // Previous loop omitted the final cumulative interval: x < iz - 1.
        for (let x = 0; x < iz; x++){
            cumulativeMSE += (MSE0[x] + MSE0[x + 1]) * (z0[x + 1] - z0[x]);
        }

        MSE0_bar[iz] = 0.5 * cumulativeMSE / (z0[iz] - z0[0]);
    }

    const int_arg = MSE0_bar.map((mse, index) =>
        -(Constants.g / (Constants.c_pd * T0[index])) * (mse - MSE0_star[index]));
    const nearestHeightIndex = height => z0.reduce((closest, sampleHeight, index) =>
        Math.abs(sampleHeight - height) < Math.abs(z0[closest] - height) ? index : closest, 0);
    const ind_LFC = nearestHeightIndex(LFC);
    const ind_EL = nearestHeightIndex(EL);
    let NCAPE = 0;

    // Python reference integrates through the interval immediately below the nearest EL index.
    for (let i = ind_LFC; i < ind_EL - 1; i++){
        const value = (0.5 * int_arg[i] + 0.5 * int_arg[i + 1]) * (z0[i + 1] - z0[i]);
        if (value === value){
            NCAPE += value;
        }
    }

    NCAPE = Math.max(0, NCAPE);
    return {NCAPE, MSE0_star, MSE0_bar};
}

/**Calculate the entraining cape of a parcel
 * @param {*} sr_wind 0-1 km AGL average storm relative wind magnitude
 * @param {*} psi Parameter defined in eqn. 52, constant for a given equilibrium level
 * @param {*} ncape Buoyancy dilution potential of the free troposphere (eqn. 54)
 * @param {*} cape Convective available potential energy (CAPE, user-defined type)
 * @returns Entraining CAPE (eqn. 55)
 */
export function calc_ecape_a(sr_wind, psi, ncape, cape){
    const ta = Math.pow(sr_wind, 2)/2;
    const tb = ((-1 - psi - (2 * psi / Math.pow(sr_wind, 2)) * ncape) / ((4 * psi) / Math.pow(sr_wind, 2)));
    const tc = (
        Math.sqrt(Math.pow((1 + psi + (2 * psi / Math.pow(sr_wind, 2)) * ncape), 2) + 8 * (psi / Math.pow(sr_wind, 2)) * (cape - (psi * ncape)))
    ) / ((4 * psi) / Math.pow(sr_wind, 2));
    var eCape = ta+tb+tc;

    return (eCape >= 0 ? eCape : 0);
}

export function calc_sr_wind(pres, uwind, vwind, hght, ifb, ift, CSMot, V){

    // var v_sr = vwind.map(row => row + CSMot.y);
    const u_sr = vwind.map(row => -row - CSMot.x);
    const v_sr = uwind.map(row => -row - CSMot.y);
    // Previous integration included segments below ifb and used a fixed 1000 m denominator.
    // var srwind = 0;
    // for (let i = 0; i < hghtAGL.length; i++){
    //     if (hghtAGL[i] > ift) break;

/** Parameter defined in eqn. 52, constant for a given equilibrium level, see COMPUTE_ECAPE.m L88 (pitchfork)
 * 
 * @param {*} el meters equilibirum lebel
 * @returns psi
 */
export function calc_psi(el){
    var sigma = 1.1;
    var alpha = 0.8;
    var l_mix = 120; //m
    var pr = 1/3;
    var ksq = 0.18;

    // return (ksq * Math.pow(alpha, 2) * Math.pow(Math.PI, 2) * l_mix) / (pr * Math.pow(sigma, 2) * el)
    return (ksq * Math.pow(alpha, 2) * Math.pow(Math.PI, 2) * l_mix) / (4 * pr * Math.pow(sigma, 2) * el)
}

 * @param {*} spechum specific humidity
 * @param {*} u_wind x comp of wind
 * @param {*} v_wind y comp of wind
 * @param {*} snding the sounding
 * @param {*} ifb 0 meters
 * @param {*} ift 1000 meters
 * @param {*} sm  right
 * @param {*} passedInfo includes {lfc, el, StormMotion:{u,v}, Cape}
 * @returns 
 */
export function calc_ecape_ncape(hght, pres, tmp, spechum, u_wind, v_wind, snding, ifb=0, ift=1000, sm="right", passedInfo){
    var el_z = passedInfo.EL;
    var lfc_z = passedInfo.LFC;

    const {NCAPE, MSE0_star, MSE0_bar} = compute_NCAPE(tmp.map(celsiusToKelvin), pres.map(row => row*100), spechum, hght, snding, 273.15, 253.15, lfc_z, el_z);
    
    const sr_wind = calc_sr_wind(pres, u_wind, v_wind, hght, ifb, ift, passedInfo.StormMotion, sm)
    const psi = calc_psi(el_z);
    const ECape = calc_ecape_a(sr_wind, psi, NCAPE, passedInfo.Cape);
    const NCape = NCAPE;

    return {ECape, NCape}
}
